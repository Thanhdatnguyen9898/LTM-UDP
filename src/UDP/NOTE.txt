-- Chuyển đổi:
    ==========================================================================================================================================
    THAO TÁC CHUYỂN ĐỔI        CÚ PHÁP CHO MẢNG ĐỐI TƯỢNG (OBJECT / WRAPPER)          CÚ PHÁP CHO MẢNG NGUYÊN THỦY (int[])
    ==========================================================================================================================================
    List -> Set                new HashSet<>(list)                                    -
    ------------------------------------------------------------------------------------------------------------------------------------------
    Set -> List                new ArrayList<>(set)                                   -
    ------------------------------------------------------------------------------------------------------------------------------------------
    Array -> List              new ArrayList<>(Arrays.asList(arr))                    Arrays.stream(arr).boxed().toList()
    ------------------------------------------------------------------------------------------------------------------------------------------
    List -> Array              list.toArray(new T[0])                                 list.stream().mapToInt(i -> i).toArray()
    ------------------------------------------------------------------------------------------------------------------------------------------
    Array -> Set               new HashSet<>(Arrays.asList(arr))                      Arrays.stream(arr).boxed().collect(Collectors.toSet())
    ------------------------------------------------------------------------------------------------------------------------------------------
    Set -> Array               set.toArray(new T[0])                                  set.stream().mapToInt(i -> i).toArray()
    ------------------------------------------------------------------------------------------------------------------------------------------
    Cắt List [a, b)            new ArrayList<>(list.subList(a, b))                    -
    ------------------------------------------------------------------------------------------------------------------------------------------
    Cắt Array [a, b)           Arrays.copyOfRange(arr, a, b)                          Arrays.copyOfRange(arr, a, b)
    ==========================================================================================================================================
